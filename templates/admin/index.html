Como Configurar a Planilha

Siga estes 4 passos simples na sua conta do Google:

  1. Abra uma planilha no Google Sheets.
  2. No menu superior, vá em Extensões → Apps Script.
  3. Apague o código padrão e cole o código abaixo.
  4. Clique em Implantar → Nova implantação:
    • Tipo: App da Web
    • Quem pode acessar: Qualquer pessoa
  5. Copie a URL do app da Web gerada e cole no campo acima!
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getActiveSheet();
    var payload = JSON.parse(e.postData.contents);

    if (payload.action === "ping") {
      return ContentService.createTextOutput(JSON.stringify({success: true, message: "Conectado!"}))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Data/Hora", "Cliente", "WhatsApp", "Cidade/UF", "Canal", "Produto", "Variação/Cor", "Qtd", "Valor Unitário (R$)", "Valor Total (R$)", "Pagamento", "Prazo", "Status", "Observações"]);
    }

    sheet.appendRow([
      payload.data_hora || new Date().toLocaleString("pt-BR"),
      payload.cliente_nome || "",
      payload.cliente_whatsapp || "",
      payload.cliente_cidade || "",
      payload.canal_venda || "WhatsApp",
      payload.produto_nome || "",
      payload.variacao_cor || "",
      Number(payload.quantidade) || 1,
      Number(payload.valor_unitario) || 0,
      Number(payload.valor_total) || 0,
      payload.forma_pagamento || "Pix",
      payload.prazo || "",
      payload.status || "Em Produção",
      payload.observacoes || ""
    ]);

    return ContentService.createTextOutput(JSON.stringify({success: true, message: "Venda gravada!"}))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
